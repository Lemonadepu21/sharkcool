cmake_minimum_required(VERSION 3.16)
project(SharkCoolDesktop VERSION 1.13.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTORCC ON)
add_compile_options("$<$<CXX_COMPILER_ID:MSVC>:/utf-8>")
find_package(Qt6 6.4 REQUIRED COMPONENTS Widgets Network Svg)
add_executable(sharkcool-window main.cpp window.cpp api_client.cpp resources.qrc)
target_link_libraries(sharkcool-window PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg)
set_target_properties(sharkcool-window PROPERTIES WIN32_EXECUTABLE TRUE)
include(CTest)
if(BUILD_TESTING)
  find_package(Qt6 REQUIRED COMPONENTS Test)
  set(CMAKE_AUTOMOC ON)
  add_executable(sharkcool-window-tests tests.cpp window.cpp api_client.cpp resources.qrc)
  target_link_libraries(sharkcool-window-tests PRIVATE Qt6::Widgets Qt6::Network Qt6::Svg Qt6::Test)
  add_test(NAME native-window COMMAND sharkcool-window-tests -platform offscreen)
endif()
add_executable(sharkcool-icon-export EXCLUDE_FROM_ALL icon_export.cpp resources.qrc)
target_link_libraries(sharkcool-icon-export PRIVATE Qt6::Gui Qt6::Svg)
option(BUILD_TRANSPORT_BRIDGE "Build Windows HID/BLE stdio adapter" ${WIN32})
if(BUILD_TRANSPORT_BRIDGE)
  find_package(Qt6 REQUIRED COMPONENTS Bluetooth)
  add_executable(sharkcool-bridge bridge/main.cpp)
  target_link_libraries(sharkcool-bridge PRIVATE Qt6::Core Qt6::Bluetooth)
  if(WIN32)
    target_link_libraries(sharkcool-bridge PRIVATE setupapi hid)
    target_compile_definitions(sharkcool-bridge PRIVATE NOMINMAX)
  endif()
  if(BUILD_TESTING)
    add_test(NAME transport-bridge-framing COMMAND sharkcool-bridge --self-test)
  endif()
endif()
if(WIN32)
  set(SHARKCOOL_ICON "${CMAKE_CURRENT_SOURCE_DIR}/../packaging/sharkcool.ico")
  configure_file(../packaging/windows/app.rc ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  target_sources(sharkcool-window PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  add_executable(SharkCool WIN32 windows_launcher.cpp ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  target_compile_definitions(SharkCool PRIVATE UNICODE _UNICODE)
  if(MINGW)
    target_link_options(SharkCool PRIVATE -municode)
  endif()
endif()
